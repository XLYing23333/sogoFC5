cmake_minimum_required(VERSION 3.16)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" SOGO_PACKAGE_VERSION LIMIT_COUNT 1)
project(sogoFC5 VERSION "${SOGO_PACKAGE_VERSION}" LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic -Werror -fstack-protector-strong)
add_compile_options("-ffile-prefix-map=${CMAKE_SOURCE_DIR}=." "-ffile-prefix-map=${CMAKE_BINARY_DIR}=build")
add_library(sogo-fc4-abi INTERFACE)
target_include_directories(sogo-fc4-abi INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/src")
find_package(Qt5 5.15 COMPONENTS Core REQUIRED)
find_package(Fcitx5Core 5.1.7 REQUIRED)
if(SOGO_SDK_ROOT)
  foreach(component Core Config Utils)
    set_property(TARGET Fcitx5::${component} PROPERTY INTERFACE_INCLUDE_DIRECTORIES "${SOGO_SDK_ROOT}/usr/include/Fcitx5/${component}")
  endforeach()
endif()
set(SOGO_VENDOR_ROOT "/usr/lib/sogo-fc5-direct/vendor/files" CACHE PATH "Installed official engine root")
set(SOGO_ENGINE_PATH "/usr/lib/sogo-fc5-direct/sogo-fc5-direct-engine" CACHE FILEPATH "Installed engine executable")
add_executable(sogo-fc5-direct-engine src/worker.cpp src/runtime_guards.cpp)
target_link_libraries(sogo-fc5-direct-engine PRIVATE sogo-fc4-abi Qt5::Core dl)
target_compile_definitions(sogo-fc5-direct-engine PRIVATE SOGO_VENDOR_ROOT="${SOGO_VENDOR_ROOT}")
set_target_properties(sogo-fc5-direct-engine PROPERTIES ENABLE_EXPORTS ON)
add_library(sogo-fc5-direct MODULE src/addon.cpp)
target_link_libraries(sogo-fc5-direct PRIVATE sogo-fc4-abi Fcitx5::Core)
target_compile_definitions(sogo-fc5-direct PRIVATE SOGO_ENGINE_PATH="${SOGO_ENGINE_PATH}" SOGO_VERSION="${PROJECT_VERSION}")
install(TARGETS sogo-fc5-direct LIBRARY DESTINATION lib/x86_64-linux-gnu/fcitx5)
install(TARGETS sogo-fc5-direct-engine RUNTIME DESTINATION lib/sogo-fc5-direct)
include(CTest)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_executable(shell-abi-test tests/shell_abi_test.cpp)
target_link_libraries(shell-abi-test PRIVATE sogo-fc4-abi)
target_compile_options(shell-abi-test PRIVATE -UNDEBUG)
add_test(NAME compiler-abi-contract COMMAND shell-abi-test)
add_test(NAME descriptor-decoder COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/descriptors_test.py")
add_executable(runtime-guards-test tests/runtime_guards_test.cpp src/runtime_guards.cpp)
target_compile_options(runtime-guards-test PRIVATE -UNDEBUG)
add_test(NAME native-runtime-guards COMMAND runtime-guards-test)
foreach(name keys queue transport profile presentation editing result_buffers scheduling latency keypad)
  add_executable(${name}-test tests/${name}_test.cpp)
  target_link_libraries(${name}-test PRIVATE sogo-fc4-abi)
  target_compile_options(${name}-test PRIVATE -UNDEBUG)
  add_test(NAME native-${name} COMMAND ${name}-test)
endforeach()
target_link_libraries(presentation-test PRIVATE Fcitx5::Core)
target_link_libraries(scheduling-test PRIVATE Fcitx5::Utils)
add_executable(presentation-benchmark tests/presentation_benchmark.cpp)
target_link_libraries(presentation-benchmark PRIVATE sogo-fc4-abi Fcitx5::Core)
find_package(Protobuf QUIET)
if(Protobuf_FOUND)
  protobuf_generate_cpp(PROTOCOL_SOURCES PROTOCOL_HEADERS
    "${CMAKE_CURRENT_SOURCE_DIR}/abi/4.2.1.145/ImeClient.proto"
    "${CMAKE_CURRENT_SOURCE_DIR}/abi/4.2.1.145/ImeServer.proto")
  add_executable(protocol-source-test tests/protocol_test.cpp ${PROTOCOL_SOURCES} ${PROTOCOL_HEADERS})
  target_include_directories(protocol-source-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(protocol-source-test PRIVATE protobuf::libprotobuf)
  target_compile_options(protocol-source-test PRIVATE -UNDEBUG)
  add_test(NAME recovered-protocol-source COMMAND protocol-source-test)
endif()
set(SOGO_FC4_ROOT "" CACHE PATH "Extracted official 4.2.1.145 package root for read-only analysis")
if(SOGO_FC4_ROOT)
  add_test(NAME official-binary-static-contract
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/static_test.py"
    "${SOGO_FC4_ROOT}")
endif()
